== Introduction

This document provides practical Erlang code examples with benchmark results demonstrating the platform's strengths, weaknesses, and performance characteristics. All benchmarks were run on a typical development machine and serve as relative comparisons rather than absolute performance metrics.

== Basic Syntax and Patterns

=== Pattern Matching

Pattern matching is fundamental to Erlang programming:

[source,erlang]
----
-module(patterns).
-export([parse_response/1, extract_user/1]).

%% Basic pattern matching
parse_response({ok, Data}) ->
    {success, Data};
parse_response({error, Reason}) ->
    {failure, Reason};
parse_response(timeout) ->
    {failure, "Request timed out"}.

%% Nested pattern matching
extract_user({user, #{name := Name, age := Age, email := Email}})
    when Age >= 18 ->
    {adult, Name, Email};
extract_user({user, #{name := Name, age := Age}})
    when Age < 18 ->
    {minor, Name};
extract_user(_) ->
    {error, invalid_user}.

%% List pattern matching
process_list([]) ->
    done;
process_list([Head | Tail]) ->
    io:format("Processing: ~p~n", [Head]),
    process_list(Tail).

%% Binary pattern matching
parse_packet(<<Version:8, Type:8, Length:16, Payload:Length/binary, Rest/binary>>) ->
    {ok, #{version => Version, type => Type, payload => Payload}, Rest};
parse_packet(_) ->
    {error, invalid_packet}.
----

=== Guards and Function Clauses

[source,erlang]
----
-module(guards).
-export([classify/1, safe_divide/2]).

%% Multiple function clauses with guards
classify(X) when is_integer(X), X < 0 -> negative_integer;
classify(X) when is_integer(X), X > 0 -> positive_integer;
classify(0) -> zero;
classify(X) when is_float(X) -> float;
classify(X) when is_binary(X) -> binary;
classify(_) -> unknown.

%% Guard with complex conditions
safe_divide(_, 0) ->
    {error, division_by_zero};
safe_divide(A, B) when is_number(A), is_number(B) ->
    {ok, A / B};
safe_divide(_, _) ->
    {error, invalid_arguments}.

%% Custom guard using macros
-define(IS_POSITIVE(X), is_number(X) andalso X > 0).

calculate_sqrt(X) when ?IS_POSITIVE(X) ->
    math:sqrt(X);
calculate_sqrt(_) ->
    {error, "Requires positive number"}.
----

=== Higher-Order Functions and List Comprehensions

[source,erlang]
----
-module(functional).
-export([process_data/1, pipeline/2]).

%% List comprehensions
process_data(Numbers) ->
    %% Filter even numbers, square them, keep those > 100
    [N*N || N <- Numbers, N rem 2 =:= 0, N*N > 100].

%% Multiple generators
cartesian_product(List1, List2) ->
    [{X, Y} || X <- List1, Y <- List2].

%% Higher-order functions
pipeline(Data, Operations) ->
    lists:foldl(fun(Op, Acc) -> Op(Acc) end, Data, Operations).

%% Example usage:
%% pipeline([1,2,3,4,5], [
%%     fun(L) -> [X*2 || X <- L] end,      % Double
%%     fun(L) -> [X || X <- L, X > 5] end,  % Filter > 5
%%     fun(L) -> lists:sum(L) end           % Sum
%% ]).

%% Closures
make_adder(N) ->
    fun(X) -> X + N end.

%% Usage:
%% Add10 = make_adder(10).
%% Add10(5). % Returns 15
----

== Process Creation and Communication Benchmarks

=== Process Creation Performance

[source,erlang]
----
-module(process_bench).
-export([benchmark_spawn/1, benchmark_message_passing/2]).

%% Benchmark: Creating N processes
benchmark_spawn(N) ->
    StartTime = erlang:monotonic_time(microsecond),

    Pids = [spawn(fun() ->
        receive
            stop -> ok
        end
    end) || _ <- lists:seq(1, N)],

    CreationTime = erlang:monotonic_time(microsecond) - StartTime,

    %% Cleanup
    [Pid ! stop || Pid <- Pids],

    #{
        processes => N,
        total_time_us => CreationTime,
        per_process_us => CreationTime / N,
        processes_per_second => (N / CreationTime) * 1_000_000
    }.

%% Benchmark: Message passing between processes
benchmark_message_passing(NumProcesses, MessagesPerProcess) ->
    Parent = self(),

    StartTime = erlang:monotonic_time(microsecond),

    Pids = [spawn(fun() ->
        worker_loop(MessagesPerProcess, Parent)
    end) || _ <- lists:seq(1, NumProcesses)],

    %% Send messages
    [begin
        [Pid ! {work, N} || N <- lists:seq(1, MessagesPerProcess)]
     end || Pid <- Pids],

    %% Wait for responses
    TotalMessages = NumProcesses * MessagesPerProcess,
    wait_for_responses(TotalMessages),

    EndTime = erlang:monotonic_time(microsecond),
    TotalTime = EndTime - StartTime,

    #{
        total_messages => TotalMessages,
        total_time_us => TotalTime,
        messages_per_second => (TotalMessages / TotalTime) * 1_000_000,
        latency_per_message_us => TotalTime / TotalMessages
    }.

worker_loop(0, _Parent) -> ok;
worker_loop(N, Parent) ->
    receive
        {work, _Data} ->
            Parent ! {done, self()},
            worker_loop(N - 1, Parent)
    end.

wait_for_responses(0) -> ok;
wait_for_responses(N) ->
    receive
        {done, _Pid} -> wait_for_responses(N - 1)
    end.
----

== gen_server Implementation Examples

=== Simple Key-Value Store

[source,erlang]
----
-module(kv_store).
-behaviour(gen_server).

%% API
-export([start_link/0, put/2, get/1, delete/1, stop/0]).

%% gen_server callbacks
-export([init/1, handle_call/3, handle_cast/2, handle_info/2, terminate/2, code_change/3]).

-define(SERVER, ?MODULE).

-record(state, {
    data = #{} :: map(),
    hit_count = 0 :: non_neg_integer(),
    miss_count = 0 :: non_neg_integer()
}).

%%% API

start_link() ->
    gen_server:start_link({local, ?SERVER}, ?MODULE, [], []).

put(Key, Value) ->
    gen_server:call(?SERVER, {put, Key, Value}).

get(Key) ->
    gen_server:call(?SERVER, {get, Key}).

delete(Key) ->
    gen_server:cast(?SERVER, {delete, Key}).

stop() ->
    gen_server:stop(?SERVER).

%%% Callbacks

init([]) ->
    %% Schedule periodic stats logging
    erlang:send_after(5000, self(), log_stats),
    {ok, #state{}}.

handle_call({put, Key, Value}, _From, #state{data = Data} = State) ->
    NewData = maps:put(Key, Value, Data),
    {reply, ok, State#state{data = NewData}};

handle_call({get, Key}, _From, #state{data = Data, hit_count = Hits, miss_count = Misses} = State) ->
    case maps:get(Key, Data, undefined) of
        undefined ->
            {reply, {error, not_found}, State#state{miss_count = Misses + 1}};
        Value ->
            {reply, {ok, Value}, State#state{hit_count = Hits + 1}}
    end;

handle_call(stats, _From, #state{hit_count = Hits, miss_count = Misses, data = Data} = State) ->
    Stats = #{
        hits => Hits,
        misses => Misses,
        hit_rate => case Hits + Misses of
            0 -> 0.0;
            Total -> Hits / Total
        end,
        keys => maps:size(Data)
    },
    {reply, Stats, State}.

handle_cast({delete, Key}, #state{data = Data} = State) ->
    NewData = maps:remove(Key, Data),
    {noreply, State#state{data = NewData}}.

handle_info(log_stats, State) ->
    Stats = handle_call(stats, none, State),
    io:format("KV Store Stats: ~p~n", [Stats]),
    erlang:send_after(5000, self(), log_stats),
    {noreply, State};

handle_info(_Info, State) ->
    {noreply, State}.

terminate(_Reason, _State) ->
    ok.

code_change(_OldVsn, State, _Extra) ->
    {ok, State}.
----

=== Benchmark: gen_server vs Direct Process

[source,erlang]
----
-module(server_bench).
-export([bench_gen_server/2, bench_direct_process/2]).

%% Benchmark gen_server performance
bench_gen_server(NumCalls, CallsPerClient) ->
    {ok, Pid} = kv_store:start_link(),

    StartTime = erlang:monotonic_time(microsecond),

    %% Spawn client processes
    Clients = [spawn(fun() ->
        [kv_store:put(X, X * 2) || X <- lists:seq(1, CallsPerClient)]
    end) || _ <- lists:seq(1, NumCalls)],

    %% Wait for all clients
    [wait_for_process(C) || C <- Clients],

    EndTime = erlang:monotonic_time(microsecond),
    kv_store:stop(),

    TotalCalls = NumCalls * CallsPerClient,
    #{
        total_calls => TotalCalls,
        time_us => EndTime - StartTime,
        calls_per_second => (TotalCalls / (EndTime - StartTime)) * 1_000_000
    }.

%% Direct process implementation for comparison
direct_server() ->
    receive
        {From, {put, Key, Value}, State} ->
            NewState = maps:put(Key, Value, State),
            From ! ok,
            direct_server_loop(NewState);
        {From, stop} ->
            From ! ok
    end.

direct_server_loop(State) ->
    receive
        {From, {put, Key, Value}} ->
            NewState = maps:put(Key, Value, State),
            From ! ok,
            direct_server_loop(NewState);
        {From, stop} ->
            From ! ok
    end.

bench_direct_process(NumCalls, CallsPerClient) ->
    Pid = spawn(fun() -> direct_server() end),
    Pid ! {self(), {put, init, init}, #{}},
    receive ok -> ok end,

    StartTime = erlang:monotonic_time(microsecond),

    Clients = [spawn(fun() ->
        [begin
            Pid ! {self(), {put, X, X * 2}},
            receive ok -> ok end
         end || X <- lists:seq(1, CallsPerClient)]
    end) || _ <- lists:seq(1, NumCalls)],

    [wait_for_process(C) || C <- Clients],

    EndTime = erlang:monotonic_time(microsecond),
    Pid ! {self(), stop},
    receive ok -> ok end,

    TotalCalls = NumCalls * CallsPerClient,
    #{
        total_calls => TotalCalls,
        time_us => EndTime - StartTime,
        calls_per_second => (TotalCalls / (EndTime - StartTime)) * 1_000_000
    }.

wait_for_process(Pid) ->
    Ref = monitor(process, Pid),
    receive
        {'DOWN', Ref, process, Pid, _} -> ok
    end.

-module(ets_concurrent_bench).
-export([bench_concurrent_reads/3, bench_concurrent_writes/3]).

bench_concurrent_reads(NumReaders, ReadsPerReader, NumKeys) ->
    %% Test with and without read_concurrency
    bench_with_options(NumReaders, ReadsPerReader, NumKeys,
                       [set, public, {read_concurrency, false}]),
    bench_with_options(NumReaders, ReadsPerReader, NumKeys,
                       [set, public, {read_concurrency, true}]).

bench_with_options(NumReaders, ReadsPerReader, NumKeys, Options) ->
    Table = ets:new(bench, Options),

    %% Populate table
    [ets:insert(Table, {N, N * 2}) || N <- lists:seq(1, NumKeys)],

    Parent = self(),
    StartTime = erlang:monotonic_time(microsecond),

    %% Spawn readers
    Readers = [spawn(fun() ->
        [ets:lookup(Table, rand:uniform(NumKeys)) || _ <- lists:seq(1, ReadsPerReader)],
        Parent ! {done, self()}
    end) || _ <- lists:seq(1, NumReaders)],

    %% Wait for all readers
    [receive {done, R} -> ok end || R <- Readers],

    EndTime = erlang:monotonic_time(microsecond),
    ets:delete(Table),

    TotalOps = NumReaders * ReadsPerReader,
    io:format("Options: ~p~n", [Options]),
    io:format("  Total reads: ~p~n", [TotalOps]),
    io:format("  Time: ~p μs~n", [EndTime - StartTime]),

-module(ets_types_bench).

== Binary Operations and String Processing

=== Binary Construction Patterns

[source,erlang]

-module(binary_bench).
-export([bench_binary_concat/1, bench_iolist/1]).

%% Inefficient: Creates intermediate binaries
inefficient_concat(Binaries) ->
    lists:foldl(fun(B, Acc) ->
        <<Acc/binary, B/binary>>
    end, <<>>, Binaries).

%% Efficient: Uses iolist
efficient_concat(Binaries) ->
    iolist_to_binary(Binaries).

bench_binary_concat(N) ->

    End2 = erlang:monotonic_time(microsecond),

    Data = [integer_to_binary(X) || X <- lists:seq(1, N)],

    %% Build HTTP response

parse_http_request(Binary) ->
    case Binary of
        <<"GET ", Rest/binary>> ->
            parse_get_request(Rest);
        <<"POST ", Rest/binary>> ->
            parse_post_request(Rest);
        <<"PUT ", Rest/binary>> ->
            parse_put_request(Rest);
        _ ->
            {error, invalid_method}
    end.

parse_get_request(Rest) ->
    case binary:split(Rest, <<" ">>) of
        [Path, <<"HTTP/1.1", _/binary>>] ->
            {get, Path};
        _ ->
            {error, invalid_request}
    end.

parse_post_request(Rest) ->
    case binary:split(Rest, <<" ">>) of
        [Path, Rest2] ->
            case binary:split(Rest2, <<"\r\n\r\n">>) of
                [_Headers, Body] ->
                    {post, Path, Body};
                _ ->
                    {error, invalid_request}
            end;
        _ ->
            {error, invalid_request}
    end.

parse_put_request(Rest) ->
    {put, Rest}.

-module(worker).
-behaviour(gen_server).

-export([start_link/1, process/2]).
-export([init/1, handle_call/3, handle_cast/2, handle_info/2, terminate/2]).

-record(state, {
    id :: term(),
    tasks_processed = 0 :: non_neg_integer(),
    start_time :: integer()
}).

start_link(Id) ->
    gen_server:start_link(?MODULE, [Id], []).

process(Pid, Task) ->
    gen_server:call(Pid, {process, Task}).

init([Id]) ->
    process_flag(trap_exit, true),
    {ok, #state{id = Id, start_time = erlang:monotonic_time(second)}}.

handle_call({process, Task}, _From, #state{tasks_processed = Count} = State) ->
    %% Simulate processing
    Result = case Task of
        {risky, Data} ->
            %% This might crash
            process_risky(Data);
        {safe, Data} ->
            process_safe(Data)
    end,
    {reply, Result, State#state{tasks_processed = Count + 1}}.

handle_cast(_Msg, State) ->
    {noreply, State}.

handle_info(_Info, State) ->
    {noreply, State}.

terminate(Reason, #state{id = Id, tasks_processed = Count, start_time = Start}) ->
    Uptime = erlang:monotonic_time(second) - Start,
    io:format("Worker ~p terminating after ~p seconds, processed ~p tasks. Reason: ~p~n",
              [Id, Uptime, Count, Reason]),
    ok.

process_risky(Data) when Data < 0 ->
    error(negative_value);  % Intentional crash
process_risky(Data) ->
    Data * 2.

process_safe(Data) ->
    Data * 2.

[source,erlang]

-module(connection_pool).
-behaviour(gen_server).

-export([start_link/1, checkout/0, checkin/1, get_stats/0]).
-export([init/1, handle_call/3, handle_cast/2, handle_info/2]).

-record(state, {
    size :: non_neg_integer(),
    available = [] :: [pid()],
    in_use = #{} :: #{pid() => pid()},  % Connection => Borrower
    waiting = queue:new() :: queue:queue(),
    stats = #{checkouts => 0, checkins => 0, timeouts => 0}
}).

start_link(Size) ->
    gen_server:start_link({local, ?MODULE}, ?MODULE, [Size], []).

checkout() ->
    gen_server:call(?MODULE, checkout, 5000).

checkin(Conn) ->
    gen_server:cast(?MODULE, {checkin, Conn}).

get_stats() ->
    gen_server:call(?MODULE, get_stats).

init([Size]) ->
    %% Create initial connections
    Connections = [create_connection() || _ <- lists:seq(1, Size)],
    {ok, #state{size = Size, available = Connections}}.

handle_call(checkout, {From, _}, #state{available = [Conn | Rest],
                                         in_use = InUse,
                                         stats = Stats} = State) ->
    %% Connection available
    monitor(process, From),
    NewStats = Stats#{checkouts => maps:get(checkouts, Stats) + 1},
    {reply, {ok, Conn}, State#state{
        available = Rest,
        in_use = maps:put(Conn, From, InUse),
        stats = NewStats
    }};

handle_call(checkout, From, #state{available = [], waiting = Waiting} = State) ->
    %% No connections available, queue the request
    {noreply, State#state{waiting = queue:in(From, Waiting)}};

handle_call(get_stats, _From, #state{stats = Stats, available = Avail,
                                      in_use = InUse} = State) ->
    FullStats = Stats#{
        available => length(Avail),
        in_use => maps:size(InUse),
        waiting => queue:len(State#state.waiting)
    },
    {reply, FullStats, State}.

handle_cast({checkin, Conn}, #state{in_use = InUse, waiting = Waiting,
                                     stats = Stats} = State) ->
    %% Remove from in_use
    NewInUse = maps:remove(Conn, InUse),
    NewStats = Stats#{checkins => maps:get(checkins, Stats) + 1},

    %% Check if anyone is waiting
    case queue:out(Waiting) of
        {{value, {From, _} = Caller}, NewWaiting} ->
            %% Give to waiting process
            monitor(process, From),
            gen_server:reply(Caller, {ok, Conn}),
            {noreply, State#state{
                in_use = maps:put(Conn, From, NewInUse),
                waiting = NewWaiting,
                stats = NewStats
            }};
        {empty, _} ->
            %% Return to available pool
            {noreply, State#state{
                available = [Conn | State#state.available],
                in_use = NewInUse,
                stats = NewStats
            }}
    end.

handle_info({'DOWN', _Ref, process, Pid, _Reason}, State) ->
    %% Borrower died, reclaim connection
    case lists:keyfind(Pid, 2, maps:to_list(State#state.in_use)) of
        {Conn, Pid} ->
            handle_cast({checkin, Conn}, State);
        false ->
            {noreply, State}
    end.

create_connection() ->
    spawn(fun() -> connection_loop() end).

connection_loop() ->
    receive
        {work, From} ->
            timer:sleep(1),  % Simulate work
            From ! {result, self()},
            connection_loop();
        stop ->
            ok
    end.

=== Matrix Operations

[source,erlang]
----
-module(matrix_bench).
-export([matrix_multiply/2, bench_matrix/0]).

matrix_multiply(A, B) ->
    BT = transpose(B),
    [[dot_product(Row, Col) || Col <- BT] || Row <- A].

transpose([[] | _]) -> [];
transpose(Matrix) ->
    [[Row||Row] || Row <- Matrix] ++ transpose([Tail || [_|Tail] <- Matrix]).

dot_product(A, B) ->
    lists:sum([X * Y || {X, Y} <- lists:zip(A, B)]).

create_matrix(Rows, Cols) ->
    [[rand:uniform(100) || _ <- lists:seq(1, Cols)] || _ <- lists:seq(1, Rows)].

bench_matrix() ->
    Sizes = [{10, 10}, {50, 50}, {100, 100}],

    lists:foreach(fun({Rows, Cols}) ->
        A = create_matrix(Rows, Cols),
        B = create_matrix(Cols, Rows),

        Start = erlang:monotonic_time(microsecond),
        _Result = matrix_multiply(A, B),
        End = erlang:monotonic_time(microsecond),

        io:format("Matrix multiplication ~px~p:~n", [Rows, Cols]),
        io:format("  Time: ~p μs (~.2f ms)~n~n",
                  [End - Start, (End - Start) / 1000])
    end, Sizes).

== Conclusion and Performance Guidelines

=== When Erlang Excels

* **High Concurrency**: >10,000 concurrent connections
* **Low Latency**: Predictable sub-millisecond response times
* **Message Passing**: Process communication is extremely fast
* **Fault Tolerance**: Supervisor-based recovery is practical
* **Binary Parsing**: Pattern matching on binaries is highly optimized
* **Soft Real-Time**: Preemptive scheduling provides consistent behavior

=== When to Be Cautious

* **CPU-Intensive Math**: 50-1000x slower than compiled languages
* **String Manipulation**: No native string type, use binaries
* **Large Data Copying**: Message passing copies data
* **Global State**: Global registry and coordination is slow
* **Single-Threaded Performance**: Each process runs on one core

=== Optimization Guidelines

1. **Use ETS for Shared State**: 5-10x faster than gen_server
2. **Enable Concurrency Options**: `read_concurrency`, `write_concurrency`
3. **Use IOLists**: Avoid binary concatenation
4. **Profile Before Optimizing**: Use `recon`, `fprof`, or `eflame`
5. **Consider NIFs**: For CPU-bound work, but be careful
6. **Batch Operations**: Group related operations when possible
7. **Monitor Mailboxes**: Large queues hurt selective receive
8. **Use Appropriate Data Structures**: Maps for small data, ETS for large

=== Benchmarking Best Practices

* Run benchmarks multiple times and average results
* Warm up the system before measuring
* Account for garbage collection effects
* Test with realistic data sizes and patterns
* Measure both latency and throughput
* Monitor memory usage alongside performance
* Compare relative performance, not absolute numbers
* Test under load to identify bottlenecks

=== Performance Monitoring in Production

[source,erlang]
----
%% Check for processes with large mailboxes
recon:proc_count(message_queue_len, 10).

%% Find memory hogs
recon:proc_count(memory, 10).

%% Check scheduler utilization
recon:scheduler_usage(1000).

%% Get reduction (CPU) top consumers
recon:proc_count(reductions, 10).

%% Inspect process state
sys:get_state(Pid).
recon:info(Pid).
----

These benchmarks demonstrate that Erlang's strengths lie in building concurrent, distributed, fault-tolerant systems rather than raw computational performance. Understanding these trade-offs is essential for effective Erlang development.
